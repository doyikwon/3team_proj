    // 넗뒪듃 뙘뾽 뿬띁
    function showToast(message, type = 'success') {
        const toast = document.getElementById('custom-toast');
        const toastContent = document.getElementById('toast-content');
        
        toast.className = `fixed bottom-5 right-5 z-50 px-6 py-4 rounded-xl shadow-2xl flex items-center gap-3 transition-all duration-300 transform translate-y-0 opacity-100 ${
            type === 'success' ? 'bg-[#1E3A2F] text-white border-l-4 border-[#10B981]' : 'bg-red-900 text-white border-l-4 border-red-500'
        }`;
        
        const icon = type === 'success' 
            ? '<i class="fa-solid fa-circle-check text-[#10B981] text-xl"></i>' 
            : '<i class="fa-solid fa-circle-exclamation text-red-500 text-xl"></i>';
            
        toastContent.innerHTML = `${icon} <span class="font-medium text-xs sm:text-sm">${message}</span>`;
        
        setTimeout(() => {
            toast.classList.add('translate-y-10', 'opacity-0');
            setTimeout(() => { toast.className = 'hidden'; }, 300);
        }, 3500);
    }

    // 酉 쟾솚 而⑦듃濡ㅻ윭
    function switchView(viewId) {
        state.currentView = viewId;
        
        // 紐⑤뱺 酉 媛由ш린
        document.querySelectorAll('.view-section').forEach(section => {
            section.classList.add('hidden');
            section.classList.remove('block');
        });
        
        // 寃 酉곕쭔 蹂댁씠湲
        const activeSection = document.getElementById(`${viewId}-view`);
        if (activeSection) {
            activeSection.classList.remove('hidden');
            activeSection.classList.add('block');
        }
        
        // 珥덇낵 吏꾨떒 솕硫댁쑝濡 吏꾩엯븷 寃쎌슦 뜲씠꽣 떎떆媛 뿰궛
        if (viewId === 'checker') {
            recalculateSupplements();
        }
        
        // 솃솕硫댁뿉 蹂듦뎄븷 닔 엳뒗 濡쒖뺄 떟蹂 씠젰씠 엳뒗吏 뙋떒븯뿬 諛곕꼫 젣뼱
        if (viewId === 'landing') {
            checkStoredAnswers();
        }
        
        window.scrollTo({ top: 0, behavior: 'smooth' });
    }

    
    // AI 留욎땄 異붿쿇 씠룞 (湲곗〈 寃곌낵 蹂댁〈)
    function navigateQuiz() {
        if (state.currentView === 'result' || state.currentView === 'quiz') return;
        
        // 1. 씠誘 댁쫰媛 吏꾪뻾 以묒씠嫄곕굹 셿猷뚮맂 긽깭씤吏 솗씤 (濡쒖뺄 state.answers 寃궗)
        if (state.answers.goals && state.answers.goals.length > 0) {
            // 씠誘 寃곌낵媛 젋뜑留 맂 긽깭
            renderRecommendationResults();
            return;
        } else if (state.quizStep > 0 && state.quizStep <= 10) {
            // 댁쫰 吏꾪뻾 以
            switchView('quiz');
            return;
        }
        
        // 2. LocalStorage 뜲씠꽣 솗씤
        const saved = localStorage.getItem('nutrimatch_survey_answers');
        if (saved) {
            loadPreviousResults();
            return;
        }
        
        // 3. 떎 뾾쑝硫 깉濡 떆옉
        startQuiz();
    }

    // 댁쫰 媛쒖떆 븿닔
    function startQuiz() {
        state.quizStep = 0; // Step 0 룞쓽遺꽣 떆옉
        state.agreements = { terms: false, age: false, sensitive: false };
        state.answers = {
            gender: '', maleConcerns: [], femaleStage: '', ageGroup: '', height: '', weight: '', bmi: null,
            exercise: [], sleep: '', diet: '', alcohol: '', caffeine: '', stress: 3,
            smoking: '', allergies: [], sideEffects: [], customSideEffect: '', conditions: [],
            goals: [],
            pillDiscomfort: '', altForm: '', preferredForm: '', buyingValues: [], portability: '', budget: '', buyingType: ''
        };
        switchView('quiz');
        renderQuizStep();
    }

    // BMI 떎떆媛 怨꾩궛湲
    function calculateBMIInline() {
        const hVal = parseFloat(document.getElementById('profile-height')?.value);
        const wVal = parseFloat(document.getElementById('profile-weight')?.value);
        const bmiResultBox = document.getElementById('bmi-calc-status');

        if (hVal && wVal && hVal > 100 && wVal > 10) {
            const hM = hVal / 100;
            const bmi = (wVal / (hM * hM)).toFixed(1);
            state.answers.height = hVal;
            state.answers.weight = wVal;
            state.answers.bmi = parseFloat(bmi);

            let category = "젙긽泥댁쨷";
            let color = "text-[#10B981]";
            if (bmi < 18.5) { category = "泥댁쨷"; color = "text-yellow-600"; }
            else if (bmi >= 23 && bmi < 25) { category = "怨쇱껜以"; color = "text-amber-600"; }
            else if (bmi >= 25) { category = "鍮꾨쭔"; color = "text-red-500 animate-pulse"; }

            if (bmiResultBox) {
                bmiResultBox.innerHTML = `
                    <div class="p-4 bg-emerald-50 rounded-2xl border border-emerald-100 flex justify-between items-center text-xs animate-fadeIn">
                        <div>
                            <span class="text-slate-400 font-medium">궛異쒕맂 BMI 吏닔</span>
                            <div class="text-lg font-bold text-[#1E3A2F] mt-0.5">BMI: <span class="${color}">${bmi}</span></div>
                        </div>
                        <span class="px-3 py-1 rounded-full bg-white font-bold text-[10px] border border-emerald-200">${category}</span>
                    </div>
                `;
            }
        } else {
            if (bmiResultBox) bmiResultBox.innerHTML = '';
            state.answers.bmi = null;
        }
    }

        }
    }

    // 蹂듭닔 꽑깮 諛 떒씪 꽑깮 媛怨듦린
    function toggleMultiSelect(arrayName, value, maxLimit = null) {
        const idx = state.answers[arrayName].indexOf(value);
        if (idx > -1) {
            state.answers[arrayName].splice(idx, 1);
        } else {
            if (maxLimit && state.answers[arrayName].length >= maxLimit) {
                showToast(`씠 빆紐⑹ 理쒕 ${maxLimit}媛쒓퉴吏 꽑깮씠 媛뒫빀땲떎.`, "error");
                return;
            }
            if (value === 'none') {
                state.answers[arrayName] = ['none'];
            } else {
                state.answers[arrayName] = state.answers[arrayName].filter(v => v !== 'none');
                state.answers[arrayName].push(value);
            }
        }
        renderQuizStep();
    }

    function setSingleSelect(variableName, value) {
        state.answers[variableName] = value;
        renderQuizStep();
    }

            "移쇱뒛": 0, "留덇렇꽕뒛": 0, "삤硫붽3": 0, "猷⑦뀒씤": 0, "떎由щ쭏由": 0
        };

        state.mySupplements.forEach(supp => {
            for (const key in supp.items) {
                if (nutrientTotals.hasOwnProperty(key)) {
                    nutrientTotals[key] += supp.items[key];
                }
            }
        });

        let warningAlerts = [];

        for (const nutrient in nutrientTotals) {
            const totalPct = nutrientTotals[nutrient];
            const rule = activeGuidelines[nutrient]; 

            if (!rule) continue;
            if (totalPct === 0) continue;

            let statusText = "쟻젙";
            let statusColor = "bg-gradient-to-r from-emerald-400 to-emerald-500";
            let alertStyle = "";

            if (totalPct < 50) {
                statusText = "遺議";
                statusColor = "bg-gradient-to-r from-slate-400 to-slate-500";
            } else if (totalPct >= rule.danger) {
                statusText = "쐞뿕 (珥덇낵)";
                statusColor = "bg-gradient-to-r from-red-500 to-rose-600 pulse-red";
                alertStyle = "text-rose-600 font-extrabold";
                warningAlerts.push({
                    nutrient: nutrient,
                    level: 'danger',
                    percent: totalPct,
                    desc: rule.desc
                });
            } else if (totalPct >= rule.maxSafe) {
                statusText = "二쇱쓽 (寃쎄퀎)";
                statusColor = "bg-gradient-to-r from-amber-400 to-amber-500";
                alertStyle = "text-amber-600 font-bold";
                warningAlerts.push({
                    nutrient: nutrient,
                    level: 'warning',
                    percent: totalPct,
                    desc: rule.desc
                });
            }

            const chartRow = document.createElement('div');
            chartRow.className = "space-y-1 animate-fadeIn";
            
            const visualProgress = Math.min(totalPct, 150); 

            chartRow.innerHTML = `
                <div class="flex justify-between items-center text-xs">
                    <span class="font-bold text-slate-700">${nutrient}</span>
                    <span class="font-mono text-[11px] ${alertStyle}">늻쟻 꽠痍: ${totalPct}${rule.unit} (${statusText})</span>
                </div>
                <div class="w-full bg-slate-100 h-5.5 rounded-full overflow-hidden relative border border-slate-200/50 shadow-inner">
                    <div class="${statusColor} h-full rounded-full flex items-center justify-end pr-3 transition-all duration-700" style="width: ${(visualProgress / 150) * 100}%">
                        <span class="text-[9px] text-white font-extrabold">${totalPct}%</span>
                    </div>
                    <div class="absolute top-0 bottom-0 w-0.5 bg-red-400 border-l border-dashed border-red-500" style="left: ${(Math.min(rule.maxSafe, 150) / 150) * 100}%" title="留욎땄 寃쎄퀎 湲곗꽑 (${rule.maxSafe}%)"></div>
                </div>
            `;
            chartArea.appendChild(chartRow);
        }

        if (warningAlerts.length === 0) {
            feedbackBox.innerHTML = `
                <div class="bg-emerald-50 border border-emerald-200 text-emerald-800 p-4 rounded-2xl text-xs flex gap-3 items-center animate-fadeIn">
                    <i class="fa-solid fa-circle-check text-xl text-emerald-500"></i>
                    <div>
                        <span class="font-bold">蹂듭슜 議고솕 긽깭: 븞쟾븿</span>
                        <p class="text-[11px] text-emerald-700 mt-1">
                            쁽옱 以묐났 꽠痍 以묒씤 쁺뼇냼뱾씠 洹븯쓽 뿰졊/紐 긽깭 鍮 뿀슜 踰붿쐞 씠궡뿉 룷吏꾪븯怨 엳뒿땲떎. 븞쟾븯寃 씪긽쟻씤 猷⑦떞쓣 怨꾩냽븯뀛룄 臾대갑빀땲떎.
                        </p>
                    </div>
                </div>
            `;
        } else {
            warningAlerts.forEach(alert => {
                const alertCard = document.createElement('div');
                const isDanger = alert.level === 'danger';
                alertCard.className = `p-4 rounded-2xl border text-xs flex gap-3 items-start animate-fadeIn ${
                    isDanger ? 'bg-red-50 border-red-200 text-red-900' : 'bg-amber-50 border-amber-200 text-amber-900'
                }`;

                alertCard.innerHTML = `
                    <i class="fa-solid ${isDanger ? 'fa-triangle-exclamation text-red-500' : 'fa-circle-exclamation text-amber-500'} text-xl mt-0.5"></i>
                    <div>
                        <span class="font-bold">${alert.nutrient} ${alert.percent}% 蹂듭슜 媛먯 (${isDanger ? '쐞뿕' : '二쇱쓽'})</span>
                        <p class="mt-1 text-[11px] ${isDanger ? 'text-red-700' : 'text-amber-800'} leading-relaxed">${alert.desc}</p>
                        <div class="mt-2.5 flex gap-2">
                            <span class="bg-white/80 px-2 py-0.5 rounded border border-slate-200 font-bold text-[9px] cursor-pointer hover:bg-white" onclick="suggestAdjustment('${alert.nutrient}', ${alert.percent})">
                                윊 뼱뼸寃 議곗젅빐빞 븷吏 븣젮二쇱꽭슂
                            </span>
                        </div>
                    </div>
                `;
                feedbackBox.appendChild(alertCard);
            });
        }
    }

    // 꽦遺 泥댄겕諛뺤뒪 넗湲 떆 씤뭼 鍮꾪솢꽦솕 젣뼱
    function toggleGridNutrient(checkbox) {
        const nutrient = checkbox.getAttribute('data-nutrient');
        const input = document.getElementById(`input-${nutrient}`);
        if (checkbox.checked) {
            input.disabled = false;
            input.classList.remove('bg-slate-100');
            input.classList.add('bg-white');
            if (!input.value) input.value = 100; // 泥댄겕 떆 湲곕낯媛 100% 엫떆 븷떦
        } else {
            input.disabled = true;
            input.classList.add('bg-slate-100');
            input.classList.remove('bg-white');
            input.value = '';
        }
    }

    // 꽦遺 泥댄겕諛뺤뒪 諛 븿웾 꽭똿 뿬띁 (寃깋/뒪罹 寃곌낵 뿰怨꾩슜)
    function populateNutrientBuilderGrid(items) {
        // 씪떒 쟾泥 珥덇린솕
        const checkboxes = document.querySelectorAll('.nutrient-chk');
        checkboxes.forEach(chk => {
            chk.checked = false;
            toggleGridNutrient(chk);
        });

        // 쟾떖諛쏆 븘씠뀥 꽭똿
        for (const nutrient in items) {
            const chk = document.querySelector(`.nutrient-chk[data-nutrient="${nutrient}"]`);
            if (chk) {
                chk.checked = true;
                toggleGridNutrient(chk);
                const input = document.getElementById(`input-${nutrient}`);
                if (input) input.value = items[nutrient];
            }
        }
    }

    // 봽由ъ뀑 쁺뼇젣 뜲씠꽣踰좎씠뒪 (珥덇린 異붽슜)
    function addPresetSupplement(key) {
        const preset = PRESET_SUPPLEMENTS[key];
        if (!preset) return;

        const isExists = state.mySupplements.some(item => item.name === preset.name);
        if (isExists) {
            showToast("씠誘 蹂닿븿뿉 벑濡앸맂 쁺뼇젣엯땲떎.", "error");
            return;
        }

        const newId = Date.now() + Math.random();
        state.mySupplements.push({
            id: newId,
            name: preset.name,

    // 而ㅼ뒪 쁺뼇젣 벑濡 泥섎━ (蹂듯빀 꽦遺 洹몃━뱶瑜 뿰궛븯뿬 떒씪 젣뭹쑝濡 理쒖쥌 媛以 벑濡)
    function addCustomSupplement() {
        const nameInput = document.getElementById('custom-supp-name');
        const name = nameInput.value.trim();

        if (!name) {
            showToast("쁺뼇젣 씠由꾩쓣 엯젰빐 二쇱꽭슂.", "error");
            return;
        }

        // 泥댄겕諛뺤뒪 洹몃━뱶뿉꽌 솢꽦솕맂 꽦遺 븿웾 씪愿 닔吏
        const checkedItems = {};
        const checkboxes = document.querySelectorAll('.nutrient-chk');
        
        checkboxes.forEach(chk => {
            if (chk.checked) {
                const nutrient = chk.getAttribute('data-nutrient');
                const input = document.getElementById(`input-${nutrient}`);
                const pct = parseInt(input.value);
                if (!isNaN(pct) && pct > 0) {
                    checkedItems[nutrient] = pct;
                }
            }
        });

        if (Object.keys(checkedItems).length === 0) {
            showToast("理쒖냼 1媛 씠긽쓽 꽦遺꾩쓣 泥댄겕븯怨 븿웾쓣 엯젰빐 二쇱꽭슂.", "error");
            return;
        }

        const newId = Date.now();
        state.mySupplements.push({
            id: newId,
            name: name,
            type: "custom_product",
            items: checkedItems
        });

        showToast(`깉 쁺뼇젣 젣뭹 "${name}" 蹂닿븿 異붽 셿猷!`, "success");

        // 엯젰뤌 諛 鍮뚮뜑 洹몃━뱶 쟾硫 珥덇린솕
        nameInput.value = '';
        checkboxes.forEach(chk => {
            chk.checked = false;
            toggleGridNutrient(chk);
        });

        const previewBox = document.getElementById('auto-ingredients-preview');
        if (previewBox) {
            previewBox.classList.add('hidden');
            previewBox.innerHTML = '';
        }

        recalculateSupplements();
    }

    // 異붿쿇 쁺뼇 꽦遺 蹂닿븿뿉 뿰룞 媛궛븯怨 吏꾨떒湲곕줈 쟾솚
    function addRecommendedToChecker() {
        let addedCount = 0;
        let activeConcerns = [...state.answers.goals];
        if (state.answers.gender === 'male' && state.answers.maleConcerns.length > 0) {
            state.answers.maleConcerns.forEach(mc => { if (!activeConcerns.includes(mc)) activeConcerns.push(mc); });
        }
        if (state.answers.femaleStage === 'pregnant' || state.answers.femaleStage === 'trying') {
            if (!activeConcerns.includes('menses')) activeConcerns.push('menses');
        }
        if (activeConcerns.length === 0) activeConcerns = ['fatigue', 'immunity'];

        activeConcerns.forEach(concern => {
            const items = INGREDIENTS_DB[concern];
            if (items) {
                items.forEach(item => {
                    // 媛긽 쁺뼇 꽦遺 異붿텧 留ㅼ묶
                    let keyNutrient = "鍮꾪誘퍫援";
                    if (item.name.includes("鍮꾪誘퍭")) keyNutrient = "鍮꾪誘퍭";
                    else if (item.name.includes("鍮꾪誘퍬")) keyNutrient = "鍮꾪誘퍬";
                    else if (item.name.includes("븘뿰")) keyNutrient = "븘뿰";
                    else if (item.name.includes("肄쒕씪寃")) keyNutrient = "鍮꾪誘퍬";
                    else if (item.name.includes("삤硫붽")) keyNutrient = "삤硫붽3";
                    else if (item.name.includes("猷⑦뀒씤")) keyNutrient = "猷⑦뀒씤";
                    else if (item.name.includes("떎由щ쭏由") || item.name.includes("諛겕뵪뒳")) keyNutrient = "떎由щ쭏由";
                    else if (item.name.includes("泥좊텇")) keyNutrient = "移쇱뒛";

                    const packName = `異붿쿇: [${item.name}]`;
                    const isExists = state.mySupplements.some(p => p.name === packName);
                    
                    if (!isExists) {
                        const itemsObj = {};
                        itemsObj[keyNutrient] = 100; // 젙웾 100% 留ㅼ묶
                        state.mySupplements.push({
                            id: Date.now() + Math.random(),
                            name: packName,
                            type: "custom_recommended",
                            items: itemsObj
                        });
                        addedCount++;
                    }
                });
            }
        });

        if (addedCount > 0) {
            showToast(`${addedCount}媛쒖쓽 異붿쿇 꽦遺꾩쓣 蹂닿븿뿉 닾엯떆궢땲떎.`, "success");
        } else {
            showToast("씠誘 異붿쿇 꽦遺꾨뱾씠 蹂닿븿뿉 諛섏쁺릺뼱 엳뒿땲떎.", "success");
        }
        
        switchView('checker');
    }
